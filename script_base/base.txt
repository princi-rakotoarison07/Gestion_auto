-- ============================================
-- TABLES DE BASE (inchangées)
-- ============================================

-- 1. TABLE CLIENT
CLIENT (
  id_client INT PRIMARY KEY AUTO_INCREMENT,
  nom VARCHAR(50) NOT NULL,
  prenom VARCHAR(50) NOT NULL,
  telephone VARCHAR(20) NOT NULL,
  email VARCHAR(100)
)

-- ============================================
-- TABLES DE RÉFÉRENCE VÉHICULES
-- ============================================

-- TABLE MARQUE (Nouveau)
MARQUE (
  id_marque INT PRIMARY KEY AUTO_INCREMENT,
  libelle VARCHAR(50) NOT NULL UNIQUE,  -- Toyota, Honda, Mazda, Renault
  pays_origine VARCHAR(50),              -- Japon, France, etc.
  actif BOOLEAN DEFAULT TRUE
)

-- TABLE MODELE (Nouveau)
MODELE (
  id_modele INT PRIMARY KEY AUTO_INCREMENT,
  id_marque INT NOT NULL,
  libelle VARCHAR(50) NOT NULL,          -- Corolla, Civic, CX-5
  type_vehicule VARCHAR(30),             -- Berline, SUV, Pick-up, etc.
  url_img VARCHAR(100),                  -- Image générique du modèle
  actif BOOLEAN DEFAULT TRUE,
  
  FOREIGN KEY (id_marque) REFERENCES MARQUE(id_marque),
  UNIQUE (id_marque, libelle)            -- Un modèle unique par marque
)

-- TABLE VEHICULE (Modifiée - représente un exemplaire physique)
VEHICULE (
  id_vehicule INT PRIMARY KEY AUTO_INCREMENT,
  id_modele INT NOT NULL,
  immatriculation VARCHAR(20) UNIQUE,
  annee INT NOT NULL CHECK (annee >= 1900 AND annee <= YEAR(CURRENT_DATE) + 1),
  
  -- Caractéristiques spécifiques à cet exemplaire
  couleur VARCHAR(30),
  kilometrage INT,
  numero_chassis VARCHAR(50) UNIQUE,
  
  -- Prix spécifiques à cet exemplaire
  prix_achat DECIMAL(10,2) NOT NULL CHECK (prix_achat > 0),
  prix_vente DECIMAL(10,2) NOT NULL CHECK (prix_vente > 0),
  
  -- Statut du véhicule
  statut VARCHAR(20) DEFAULT 'en_stock',  -- en_stock, vendu, en_réparation
  date_acquisition DATE DEFAULT CURRENT_DATE,
  
  FOREIGN KEY (id_modele) REFERENCES MODELE(id_modele),
  
  CONSTRAINT chk_statut_vehicule 
    CHECK (statut IN ('en_stock', 'vendu', 'en_reparation', 'reserve'))
)
-- 3. TABLE EMPLOYE
EMPLOYE (
  id_employe INT PRIMARY KEY AUTO_INCREMENT,
  nom VARCHAR(50) NOT NULL,
  prenom VARCHAR(50) NOT NULL,
  telephone VARCHAR(20)
)

-- ============================================
-- TABLES DE RÉFÉRENCE
-- ============================================

-- 4. TABLE MODE_PAIEMENT
MODE_PAIEMENT (
  id_mode_paiement INT PRIMARY KEY AUTO_INCREMENT,
  libelle VARCHAR(50) NOT NULL UNIQUE,  -- espece, carte, cheque, Mvola, Orange
  actif BOOLEAN DEFAULT TRUE             -- permet de désactiver sans supprimer
)

-- 5. TABLE STATUT_PAIEMENT (NOUVELLE)
STATUT_PAIEMENT (
  id_statut_paiement INT PRIMARY KEY AUTO_INCREMENT,
  libelle VARCHAR(50) NOT NULL UNIQUE,  -- en_attente, partiel, complet, retard
  actif BOOLEAN DEFAULT TRUE
)

-- ============================================
-- TABLES VENTE ET PAIEMENT MODIFIÉES
-- ============================================

-- 6. TABLE VENTE (avec référence au statut)
VENTE (
  id_vente INT PRIMARY KEY AUTO_INCREMENT,
  date_vente DATE NOT NULL DEFAULT CURRENT_DATE,
  id_vehicule INT NOT NULL,
  id_client INT NOT NULL,
  id_employe INT NOT NULL,
  montant_total DECIMAL(10,2) NOT NULL CHECK (montant_total > 0),
  
  -- NOUVEAU : Référence à la table STATUT_PAIEMENT
  id_statut_paiement INT NOT NULL DEFAULT 1,  -- 1 = 'en_attente' par défaut
  date_paiement_complet DATE,  -- Quand le paiement est terminé
  
  -- Contrainte unique : un véhicule ne peut être vendu qu'une fois
  UNIQUE (id_vehicule),
  
  -- Clés étrangères
  FOREIGN KEY (id_vehicule) REFERENCES VEHICULE(id_vehicule),
  FOREIGN KEY (id_client) REFERENCES CLIENT(id_client),
  FOREIGN KEY (id_employe) REFERENCES EMPLOYE(id_employe),
  FOREIGN KEY (id_statut_paiement) REFERENCES STATUT_PAIEMENT(id_statut_paiement)
)

-- 7. TABLE PAIEMENT
PAIEMENT (
  id_paiement INT PRIMARY KEY AUTO_INCREMENT,
  id_vente INT NOT NULL,
  id_mode_paiement INT NOT NULL,
  date_paiement DATE NOT NULL DEFAULT CURRENT_DATE,
  montant DECIMAL(10,2) NOT NULL CHECK (montant > 0),
  reference VARCHAR(100),  -- numéro de chèque, transaction, etc.
  
  FOREIGN KEY (id_vente) REFERENCES VENTE(id_vente),
  FOREIGN KEY (id_mode_paiement) REFERENCES MODE_PAIEMENT(id_mode_paiement)
)

-- 8. TABLE CAISSE
CAISSE (
  id_caisse INT PRIMARY KEY AUTO_INCREMENT,
  solde DECIMAL(10,2) DEFAULT 0,
  date_derniere_operation DATETIME DEFAULT CURRENT_TIMESTAMP
)

-- 9. TABLE OPERATION_CAISSE
OPERATION_CAISSE (
  id_operation INT PRIMARY KEY AUTO_INCREMENT,
  id_caisse INT NOT NULL,
  type_operation VARCHAR(10) NOT NULL CHECK (type_operation IN ('entree', 'sortie')),
  montant DECIMAL(10,2) NOT NULL CHECK (montant > 0),
  date_operation DATETIME DEFAULT CURRENT_TIMESTAMP,
  description VARCHAR(255),
  id_paiement INT,
  
  FOREIGN KEY (id_caisse) REFERENCES CAISSE(id_caisse),
  FOREIGN KEY (id_paiement) REFERENCES PAIEMENT(id_paiement)
)
